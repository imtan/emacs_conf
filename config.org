*** 画像の貼り付け
OSXやLinuxを使用する場合はpngpasteをインストールする必要がある。
#+begin_src emacs-lisp :tangle yes
  ;; 単独行の画像は中央揃えで表示（olivettiの中央カラムに合わせる）
  ;; 個別に変えたい場合は #+ATTR_ORG: :align left で上書き
  (setq org-image-align 'center)

  (if IS-WINDOWS
      (with-eval-after-load 'org
        (defun org-insert-clipboard-image ()
    	"Clipboardの画像を ./img/YYYY-mm-dd-HH-MM-SS.png に保存してリンクを挿入。"
    	(interactive)
    	(let* ((filename (concat (format-time-string "img/%Y-%m-%d-%H-%M-%S") ".png"))
                 (script   (expand-file-name "~/.emacs.d/windows/paste_image.ps1"))
                 (_ (let ((dir (file-name-directory filename)))
                      (unless (file-directory-p dir) (make-directory dir t))))
                 (exit-code
    		(call-process
    		 "powershell" nil nil nil
    		 "-NoProfile" "-NonInteractive" "-ExecutionPolicy" "Bypass" "-STA"
    		 "-File" script
    		 "-FileName" filename)))
    	  (if (and (integerp exit-code) (= exit-code 0) (file-exists-p filename))
                (progn
    		(insert (format "[[file:%s]]" filename))
    		(org-display-inline-images))
              (user-error "画像の保存に失敗しました (exit=%s)" exit-code))))
        (define-key org-mode-map (kbd "C-M-y") #'org-insert-clipboard-image)))

  ;; macOSでクリップボード画像を ./img/YYYY-mm-dd-HH-MM-SS.png に保存して貼り付け
  (when IS-MAC
    (with-eval-after-load 'org
      (defun org-insert-clipboard-image ()
        "Clipboardの画像を ./img/YYYY-mm-dd-HH-MM-SS.png に保存してリンクを挿入。"
        (interactive)
        (unless (executable-find "pngpaste")
          (user-error "pngpaste が見つかりません。`brew install pngpaste` で入れてください"))
        (let* ((filename (concat (format-time-string "img/%Y-%m-%d-%H-%M-%S") ".png"))
               (dir (file-name-directory filename)))
          (unless (file-directory-p dir)
            (make-directory dir t))
          (let ((exit-code (call-process "pngpaste" nil nil nil filename)))
            (if (and (integerp exit-code) (= exit-code 0) (file-exists-p filename))
                (progn
                  (insert (format "[[file:%s]]" filename))
                  ;; 既存のインライン画像を更新
                  (org-display-inline-images))
              (user-error "画像の保存に失敗しました (exit=%s). クリップボードに画像がありますか？" exit-code)))))
      (define-key org-mode-map (kbd "C-M-y") #'org-insert-clipboard-image)))
#+end_src

*** Eagleから画像挿入
Eagle(eagle.cool)のローカルAPI経由で画像を選び、ライブラリ実体への
~[[file:~/Dropbox/...]]~ リンクを挿入する（コピーはしない）。
チルダ表記なので、Dropboxがホーム直下にあるOSならどこでも同じリンクで表示できる。
Eagleが起動している必要がある。
キーバインドは org-mode 内で ~C-M-e~（キーワード検索）、~C-c e~（フォルダから選択）。
#+begin_src emacs-lisp :tangle yes
  (defvar my/eagle-api-base "http://localhost:41595"
    "Eagle ローカルAPI のベースURL。")

  (defvar my/eagle-image-exts
    '("png" "jpg" "jpeg" "gif" "webp" "bmp" "svg" "avif")
    "orgに挿入する対象とする画像拡張子。")

  (defun my/eagle--get (path)
    "Eagle API の PATH へGETし、JSONの data 部を alist で返す。"
    (let ((buf (url-retrieve-synchronously (concat my/eagle-api-base path) t t 8)))
      (unless buf (user-error "Eagle API に接続できません（Eagleは起動中？）"))
      (with-current-buffer buf
        (goto-char (or (bound-and-true-p url-http-end-of-headers) (point-min)))
        (let* ((raw (buffer-substring-no-properties (point) (point-max)))
               (str (decode-coding-string raw 'utf-8))
               (json-object-type 'alist)
               (json-array-type 'list)
               (json-key-type 'symbol)
               (obj (json-read-from-string str)))
          (kill-buffer buf)
          (unless (equal (alist-get 'status obj) "success")
            (user-error "Eagle API エラー: %s" (alist-get 'status obj)))
          (alist-get 'data obj)))))

  (defun my/eagle--library-path ()
    "現在のEagleライブラリのディレクトリパス。"
    (let ((d (my/eagle--get "/api/library/info")))
      (or (alist-get 'path (alist-get 'library d))
          (alist-get 'path d))))

  (defvar my/eagle--preview-buffer "*Eagle Preview*"
    "Eagle画像プレビュー用バッファ名。")

  (defvar my/eagle-preview-max-size 300
    "Eagleプレビュー画像の最大ピクセル(幅/高さ)。小さいほど高速。")

  (defun my/eagle--preview-path (it lib)
    "アイテム IT のプレビュー用パス（サムネイル優先、無ければ実体）。"
    (let* ((id (alist-get 'id it))
           (name (alist-get 'name it))
           (ext (alist-get 'ext it))
           (thumb (format "%s/images/%s.info/%s_thumbnail.png" lib id name))
           (orig (format "%s/images/%s.info/%s.%s" lib id name ext)))
      (if (file-exists-p thumb) thumb orig)))

  (defun my/eagle--orig-path (it lib)
    "アイテム IT の実体（オリジナル）パス。"
    (format "%s/images/%s.info/%s.%s"
            lib (alist-get 'id it) (alist-get 'name it) (alist-get 'ext it)))

  (defun my/eagle--show-preview (path)
    "PATH の画像を右側のサイドウィンドウに表示する。"
    (when (and path (file-exists-p path) (display-graphic-p))
      (let ((buf (get-buffer-create my/eagle--preview-buffer)))
        (with-current-buffer buf
          (let ((inhibit-read-only t))
            (erase-buffer)
            (insert-image (create-image path nil nil
                                        :max-width my/eagle-preview-max-size
                                        :max-height my/eagle-preview-max-size))
            (goto-char (point-min)))
          (setq-local cursor-type nil)
          (setq-local mode-line-format nil)
          (setq buffer-read-only t))
        (display-buffer buf
                        '((display-buffer-in-side-window)
                          (side . right) (slot . 0) (window-width . 0.45))))))

  (defun my/eagle--kill-preview ()
    "プレビューのバッファとウィンドウを片付ける。"
    (let ((buf (get-buffer my/eagle--preview-buffer)))
      (when buf
        (let ((win (get-buffer-window buf)))
          (when (window-live-p win) (ignore-errors (delete-window win))))
        (kill-buffer buf))))

  (defun my/eagle--filter-images (items)
    "ITEMS から画像拡張子のものだけを返す。"
    (seq-filter
     (lambda (it)
       (member (downcase (or (alist-get 'ext it) "")) my/eagle-image-exts))
     items))

  (defun my/eagle--pick-and-insert (items lib)
    "ITEMS からプレビュー付きで1枚選び、実体へのorgリンクを挿入する。"
    (unless items (user-error "該当する画像アイテムがありません"))
    (let* ((cands
              (mapcar
               (lambda (it)
                 (let* ((name (alist-get 'name it))
                        (ext  (alist-get 'ext it))
                        (tags (alist-get 'tags it))
                        (label (format "%s.%s%s" name ext
                                       (if tags
                                           (format "  [%s]"
                                                   (mapconcat #'identity
                                                              (seq-take tags 5) ", "))
                                         ""))))
                   (cons label it)))
               items))
             (choice
              (unwind-protect
                  (consult--read
                   (mapcar #'car cands)
                   :prompt "Eagle画像(移動でプレビュー): "
                   :require-match t
                   :sort nil
                   :category 'file
                   :state (lambda (action cand)
                            (pcase action
                              ('preview
                               (when cand
                                 (my/eagle--show-preview
                                  (my/eagle--preview-path
                                   (cdr (assoc cand cands)) lib))))
                              ((or 'return 'exit) (my/eagle--kill-preview)))))
                (my/eagle--kill-preview)))
             (it (cdr (assoc choice cands)))
             (src (my/eagle--orig-path it lib)))
        (unless (file-exists-p src)
          (user-error "Eagleの実体が見つかりません: %s" src))
        ;; ライブラリ(Dropbox内)の実体へ直接リンクする。abbreviate-file-name で
        ;; ~/Dropbox/... 形式になるため、Dropboxがホーム直下にあるOSなら共通で表示できる
        (let ((link (abbreviate-file-name (expand-file-name src))))
          (insert (format "[[file:%s]]" link))
          (org-display-inline-images)
          (message "Eagleから挿入: %s" link))))

  (defun org-insert-eagle-image (&optional keyword)
    "Eagleから画像をプレビューしながら選び、ライブラリ実体へのorgリンクを挿入する。
  リンクは ~/Dropbox/... のチルダ形式で書くため、OSをまたいで表示できる。
  KEYWORD を与えると名前/タグで絞り込む。空なら最近順。"
    (interactive (list (read-string "Eagle 検索キーワード(空=最近順): ")))
    (require 'consult)
    (let ((q (if (and keyword (> (length keyword) 0))
                 (format "/api/item/list?limit=200&orderBy=-MTIME&keyword=%s"
                         (url-hexify-string keyword))
               "/api/item/list?limit=200&orderBy=-MTIME")))
      (my/eagle--pick-and-insert
       (my/eagle--filter-images (my/eagle--get q))
       (my/eagle--library-path))))

  (defun my/eagle--flatten-folders (folders &optional prefix)
    "FOLDERS の階層を「親/子」ラベルの (LABEL . ID) リストに平坦化する。"
    (mapcan (lambda (f)
              (let* ((name (alist-get 'name f))
                     (label (if prefix (concat prefix "/" name) name)))
                (cons (cons label (alist-get 'id f))
                      (my/eagle--flatten-folders (alist-get 'children f) label))))
            folders))

  (defun org-insert-eagle-image-from-folder ()
    "Eagleのフォルダを選び、その中の画像からorgリンクを挿入する。"
    (interactive)
    (require 'consult)
    (let* ((folders (my/eagle--flatten-folders (my/eagle--get "/api/folder/list")))
           (choice (progn
                     (unless folders (user-error "Eagleにフォルダがありません"))
                     (completing-read "Eagleフォルダ: " (mapcar #'car folders) nil t)))
           (id (cdr (assoc choice folders))))
      (my/eagle--pick-and-insert
       (my/eagle--filter-images
        (my/eagle--get
         (format "/api/item/list?limit=200&orderBy=-MTIME&folders=%s" id)))
       (my/eagle--library-path))))

  (with-eval-after-load 'org
    (define-key org-mode-map (kbd "C-M-e") #'org-insert-eagle-image)
    (define-key org-mode-map (kbd "C-c e") #'org-insert-eagle-image-from-folder))
#+end_src

*** 本のキャプチャ（ISBN自動取得）
~C-c c B~ でISBN（またはタイトル）を入れると、openBD → Google Books の順で
書誌情報を取得して ~books/books.org~ の先頭にエントリを作る。
#+begin_src emacs-lisp :tangle yes
  (defun my/book--http-json (url)
    "URL からJSONを取得してパースする。失敗したら nil。"
    (ignore-errors
      (let ((buf (url-retrieve-synchronously url t t 10)))
        (when buf
          (with-current-buffer buf
            (goto-char (or (bound-and-true-p url-http-end-of-headers) (point-min)))
            (let ((raw (buffer-substring-no-properties (point) (point-max)))
                  (json-object-type 'alist)
                  (json-array-type 'list)
                  (json-key-type 'symbol))
              (prog1 (json-read-from-string (decode-coding-string raw 'utf-8))
                (kill-buffer buf))))))))

  (defun my/book--fmt-date (d)
    "openBD の出版日(YYYYMMDD/YYYYMM)を YYYY-MM-DD / YYYY-MM に整える。"
    (cond ((null d) nil)
          ((string-match "\\`\\([0-9]\\{4\\}\\)\\([0-9]\\{2\\}\\)\\([0-9]\\{2\\}\\)?\\'" d)
           (mapconcat #'identity
                      (delq nil (list (match-string 1 d) (match-string 2 d)
                                      (match-string 3 d)))
                      "-"))
          (t d)))

  (defun my/book--fetch-openbd (isbn)
    "openBD から ISBN の書誌情報を取得する。見つからなければ nil。"
    (let* ((data (my/book--http-json (concat "https://api.openbd.jp/v1/get?isbn=" isbn)))
           (s (alist-get 'summary (car data))))
      (when (and s (not (string-empty-p (or (alist-get 'title s) ""))))
        `((title . ,(alist-get 'title s))
          (author . ,(alist-get 'author s))
          (publisher . ,(alist-get 'publisher s))
          (pubdate . ,(my/book--fmt-date (alist-get 'pubdate s)))
          (isbn . ,(alist-get 'isbn s))
          (cover . ,(alist-get 'cover s))))))

  (defun my/book--fetch-openlibrary-isbn (isbn)
    "OpenLibrary から ISBN の書誌情報を取得する。見つからなければ nil。
  Google Books APIはキー無しの無料枠が0のため、洋書はこちらを使う。"
    (let* ((data (my/book--http-json
                  (format "https://openlibrary.org/api/books?bibkeys=ISBN:%s&format=json&jscmd=data" isbn)))
           (b (alist-get (intern (concat "ISBN:" isbn)) data)))
      (when b
        `((title . ,(alist-get 'title b))
          (author . ,(mapconcat (lambda (a) (alist-get 'name a))
                                (alist-get 'authors b) ", "))
          (publisher . ,(mapconcat (lambda (p) (alist-get 'name p))
                                   (alist-get 'publishers b) ", "))
          (pubdate . ,(alist-get 'publish_date b))
          (isbn . ,isbn)
          (cover . ,(alist-get 'medium (alist-get 'cover b)))))))

  (defun my/book--search-openlibrary (q)
    "OpenLibrary をタイトル Q で検索し、書誌 alist のリストを返す。"
    (let ((data (my/book--http-json
                 (concat "https://openlibrary.org/search.json?limit=10"
                         "&fields=title,author_name,publisher,first_publish_year,isbn"
                         "&title=" (url-hexify-string q)))))
      (mapcar (lambda (d)
                `((title . ,(alist-get 'title d))
                  (author . ,(mapconcat #'identity (alist-get 'author_name d) ", "))
                  (publisher . ,(car (alist-get 'publisher d)))
                  (pubdate . ,(let ((y (alist-get 'first_publish_year d)))
                                (and y (number-to-string y))))
                  (isbn . ,(seq-find (lambda (i) (= (length i) 13))
                                     (alist-get 'isbn d)))))
              (alist-get 'docs data))))

  (defun my/book--http-xml (url)
    "URL からXMLを取得してパースする。失敗したら nil。"
    (require 'xml)
    (require 'dom)
    (ignore-errors
      (let ((buf (url-retrieve-synchronously url t t 10)))
        (when buf
          (with-current-buffer buf
            (goto-char (or (bound-and-true-p url-http-end-of-headers) (point-min)))
            (let ((raw (buffer-substring-no-properties (point) (point-max))))
              (kill-buffer buf)
              (with-temp-buffer
                (insert (decode-coding-string raw 'utf-8))
                (car (xml-parse-region (point-min) (point-max))))))))))

  (defun my/book--search-ndl (q)
    "NDLサーチ(国会図書館)を Q で検索し、書誌 alist のリストを返す。
  any=全項目検索＋mediatype=1(図書のみ)で再現率とノイズのバランスを取る。"
    (let ((dom (my/book--http-xml
                (concat "https://ndlsearch.ndl.go.jp/api/opensearch?cnt=20&mediatype=books&any="
                        (url-hexify-string q)))))
      (delq nil
            (mapcar
             (lambda (item)
               (let ((title (dom-text (car (dom-by-tag item 'title))))
                     (author (dom-text (car (dom-by-tag item 'author))))
                     (publisher (dom-text (car (dom-by-tag item 'dc:publisher))))
                     (issued (dom-text (car (dom-by-tag item 'dcterms:issued))))
                     (isbn (car (delq nil
                                      (mapcar
                                       (lambda (id)
                                         (when (equal (dom-attr id 'xsi:type) "dcndl:ISBN")
                                           (replace-regexp-in-string "-" "" (dom-text id))))
                                       (dom-by-tag item 'dc:identifier))))))
                 (when (and title (not (string-empty-p title)))
                   `((title . ,title) (author . ,author)
                     (publisher . ,publisher) (pubdate . ,issued)
                     (isbn . ,isbn)))))
             (dom-by-tag dom 'item)))))

  (defun my/book--dedup-sort (cands q)
    "書誌候補 CANDS を重複排除し、Q をタイトルに含むものを上位に並べる。
  重複キーは ISBN、無ければタイトル+著者。"
    (let (seen out)
      (dolist (b cands)
        (let ((key (or (alist-get 'isbn b)
                       (concat (alist-get 'title b) "/" (or (alist-get 'author b) "")))))
          (unless (member key seen)
            (push key seen)
            (push b out))))
      (sort (nreverse out)
            (lambda (a b) (> (my/book--score a q) (my/book--score b q))))))

  (defun my/book--score (b q)
    "候補 B のクエリ Q への一致度スコア。
  各語: タイトル一致+2/著者一致+1、先頭語のタイトル前方一致は+4。"
    (let ((title (or (alist-get 'title b) ""))
          (author (or (alist-get 'author b) ""))
          (tokens (split-string q "[ 　]+" t))
          (s 0))
      (when (and tokens (string-prefix-p (car tokens) title))
        (setq s (+ s 4)))
      (dolist (tok tokens s)
        (when (string-match-p (regexp-quote tok) title) (setq s (+ s 2)))
        (when (string-match-p (regexp-quote tok) author) (setq s (+ s 1))))))

  (defun my/book--select (cands)
    "書誌候補 CANDS の一覧から1冊選んで alist を返す。"
    (let* ((table (mapcar (lambda (b)
                            (cons (format "%s / %s / %s%s"
                                          (alist-get 'title b)
                                          (or (alist-get 'author b) "?")
                                          (or (alist-get 'publisher b) "?")
                                          (let ((d (alist-get 'pubdate b)))
                                            (if d (format " (%s)" d) "")))
                                  b))
                          cands))
           (choice (completing-read "候補から選択: " (mapcar #'car table) nil t)))
      (cdr (assoc choice table))))

  (defun my/book--fetch (query)
    "QUERY(ISBNまたはタイトル)から書誌情報の alist を返す。
  ISBNなら openBD → OpenLibrary の順で即確定。
  タイトルなら NDL → OpenLibrary で検索し、候補一覧から選択する。"
    (let* ((isbn (replace-regexp-in-string "[- 　]" "" query))
           (isbn-p (string-match-p "\\`[0-9]\\{9,12\\}[0-9Xx]\\'" isbn)))
      (if isbn-p
          (or (my/book--fetch-openbd isbn)
              (my/book--fetch-openlibrary-isbn isbn)
              (user-error "書誌情報が見つかりません: %s" query))
        (let ((cands (or (my/book--search-ndl query)
                         (my/book--search-openlibrary query))))
          (unless cands (user-error "書誌情報が見つかりません: %s" query))
          (my/book--select (my/book--dedup-sort cands query))))))

  (defun my/book--clean (s)
    "改行・タブを潰して前後の空白を落とす。"

  (defun my/book--prop (key val)
    "VAL が空でなければプロパティ行の文字列を返す。"

      (when (and v (not (string-empty-p v)))

       ":経費: No\n"

       (my/book--prop "出版社" (alist-get 'publisher b))
       (my/book--prop "出版日" (alist-get 'pubdate b))

  (defun my/claude-code--buf-name ()
    "プロジェクトに対応するClaude Codeバッファ名を返す."
    (let ((root (or (when (project-current)
                      (project-root (project-current)))
                    default-directory)))
      (format "*claude-code: %s*"
              (file-name-nondirectory
               (directory-file-name root)))))

  (defun my/claude-code--root ()
    "プロジェクトルートまたはカレントディレクトリを返す."
    (or (when (project-current)
          (project-root (project-current)))
        default-directory))

  (defun my/claude-code ()
    "現在のプロジェクトルートでClaude Codeを起動する.
  プロジェクト外の場合はカレントディレクトリで起動."
    (interactive)
    (let ((buf-name (my/claude-code--buf-name)))
      (unless (get-buffer buf-name)
        (let ((default-directory (my/claude-code--root)))
          (vterm buf-name)
          (vterm-send-string "claude\n")))
      (my/claude-code--setup-window buf-name)))

  (defun my/claude-code--setup-window (buf-name)
    "Claude Codeバッファをサイドウィンドウに表示する."
    (when-let ((buf (get-buffer buf-name)))
      (display-buffer
       buf
       '(display-buffer-in-side-window
         (side . right)
         (window-width . 0.42)
         (slot . 0)))))

  (defun my/claude-code-toggle ()
    "Claude Codeサイドウィンドウをトグル表示する."
    (interactive)
    (let* ((buf-name (my/claude-code--buf-name))
           (win (get-buffer-window buf-name)))
      (if win
          (delete-window win)
        (my/claude-code))))

  (defun my/claude-code-new-session ()
    "新しいClaude Codeセッションを強制的に起動する."
    (interactive)
    (let* ((buf-name (my/claude-code--buf-name))
           (root (my/claude-code--root)))
      (when (get-buffer buf-name)
        (let ((kill-buffer-query-functions nil))
          (kill-buffer buf-name)))
      (let ((default-directory root))
        (vterm buf-name)
        (let ((buf buf-name))
          (run-at-time 0.5 nil
                       `(lambda ()
                          (when (buffer-live-p (get-buffer ,buf))
                            (with-current-buffer ,buf
                              (vterm-send-string "claude\n")))))))
      (my/claude-code--setup-window buf-name)))

  (defun my/claude-code-send-region (start end)
    "選択リージョンをClaude Codeに送信する.
  Claude Codeが起動していない場合は先に起動する."
    (interactive "r")
    (let ((text (buffer-substring-no-properties start end))
          (buf-name (my/claude-code--buf-name)))
      (unless (get-buffer buf-name)
        (my/claude-code)
        (let ((waited 0))
          (while (and (not (get-buffer buf-name)) (< waited 5))
            (sit-for 0.5)
            (setq waited (+ waited 0.5)))))
      (if (get-buffer buf-name)
          (progn
            (with-current-buffer buf-name
              (vterm-send-string text)
              (vterm-send-return))
            (my/claude-code--setup-window buf-name))
        (error "Claude Codeバッファ '%s' の起動に失敗しました" buf-name))))

    "現在のファイルパスをClaude Codeに送信する."

          (my/claude-code)
          (let ((waited 0))
            (while (and (not (get-buffer buf-name)) (< waited 5))
              (sit-for 0.5)
              (setq waited (+ waited 0.5)))))
        (if (get-buffer buf-name)
            (progn
              (with-current-buffer buf-name
                (vterm-send-string (concat "このファイルを見てください: " rel-path))
                (vterm-send-return))
              (my/claude-code--setup-window buf-name))
          (error "Claude Codeバッファ '%s' の起動に失敗しました" buf-name)))))

フック本体は =~/.local/bin/ai-ide-hook.sh=、登録は =~/.claude/settings.json= と =~/.codex/hooks.json=。
spoolはDropboxの外に置くこと（同期書き戻しでfile-notifyが二重発火する）。

=C-c i= でチートシート付きメニュー（hydra）が開く。キーは覚えなくてよい。

| キーバインド | 動作                                         |
|--------------+----------------------------------------------|
| C-c i c      | checkpoint（エージェント実行前に打つ）       |
| C-c i d      | checkpointからの差分（diff-mode・構文色付き）|
| C-c i D      | このファイルだけcheckpointとediff            |
| C-c i j      | digestを日報に追記                           |
| C-c i n / p  | AIが書いた箇所を巡回                         |
| C-c i k      | overlayを消す（C-uで全バッファ）             |
| C-c i l      | イベント一覧                                 |
| C-c i r      | 未保存バッファとの衝突をediffで解決          |

#+begin_src emacs-lisp :tangle yes
  ;; ~/.emacs.d はload-path外のため直接ロードする。
  ;; プレフィクスは C-c i（C-c a はorg-agendaで使用中）
  (load (expand-file-name "ai-ide.el" user-emacs-directory) t 'nomessage)
  (when (fboundp 'ai-ide-mode)
    (ai-ide-mode 1)

    ;; C-c i でワークフローのチートシートを兼ねたメニューを開く。
    ;; キーは ai-ide-command-map と同じ並びなので直打ちの記憶もそのまま使える
    (defhydra hydra-ai-ide (:color blue :hint nil)
      "
 AI編集の観測   流れ: _c_ → エージェントに作業させる → _n_/_p_ で巡回 → _d_ で差分 → _j_ で日報 → _k_

  _c_: checkpoint を切る（実行前に・完了通知を待つ）   _n_: 次のAI編集箇所へ
  _d_: checkpoint からの差分を通し読み                 _p_: 前のAI編集箇所へ
  _D_: このファイルだけ ediff                          _k_: ハイライト消去（C-u で全バッファ）

      ("d" ai-ide-diff)
      ("D" ai-ide-diff-file)
      ("j" ai-ide-journal)

      ("p" ai-ide-prev-mark :color red)
      ("k" ai-ide-clear-marks)

  (defun my/gemini--replace-region-with-response (prompt start end)
    "リージョン(START END)のテキストをPROMPTでGeminiに送り、結果でその場置換する."
    (let ((text (buffer-substring-no-properties start end))
          (buf  (current-buffer))
          (m-start (copy-marker start))
          (m-end   (copy-marker end)))
      (message "Gemini に送信中...")
      (gptel-request
       (concat prompt "\n\n" text)
       :buffer buf
       :callback
       `(lambda (response info)
          (cond
           ((not response)
            (message "Gemini エラー: %s" (plist-get info :status)))
           ((not (stringp response))
            nil) ; reasoning ブロックはスキップ
           (t
            (with-current-buffer (plist-get info :buffer)
              (let ((inhibit-read-only t))
                (delete-region ,m-start ,m-end)
                (goto-char ,m-start)
                (insert response)
                (set-marker ,m-start nil)
                (set-marker ,m-end nil)))
            (message "完了")))))))

  (defun my/gemini--insert-after-region (prompt start end)
    "リージョンのテキストをGeminiに送り、結果をリージョン直後に挿入する."

          (buf  (current-buffer))
          (pos  (copy-marker end)))
      (message "Gemini に送信中...")
      (gptel-request
       (concat prompt "\n\n" text)
       :buffer buf
       :callback
       `(lambda (response info)
          (cond
           ((not response)
            (message "Gemini エラー: %s" (plist-get info :status)))
           ((not (stringp response))
            nil) ; reasoning ブロックはスキップ
           (t
            (with-current-buffer (plist-get info :buffer)
              (goto-char ,pos)
              (insert "\n\n" response)
              (set-marker ,pos nil))
            (message "完了")))))))

  (defun my/gemini-draft-blog-from-notes (start end)
    "選択リージョンのメモ・箇条書きからブログ記事の下書きを生成し、直後に挿入する."

  - org-mode形式（見出しは * や ** を使う）で出力する
  - 導入・本文・まとめの構成にする
  - コード例があれば #+begin_src ... #+end_src で囲む"
     start end))

  (defun my/gemini-draft-blog-interactive ()
    "テーマを入力してブログ記事の下書きを新しいバッファに生成する."
    (interactive)
    (let* ((theme (read-string "ブログテーマ: "))
           (lang  (completing-read "言語: " '("日本語" "English") nil t "日本語"))
           (prompt (if (string= lang "日本語")
                       (format "「%s」というテーマで技術ブログ記事の下書きを書いてください。
  org-mode形式（* 見出し, ** サブ見出し）で、
  導入・背景・本文・コード例・まとめの構成で出力してください。" theme)
                     (format "Write a draft blog post about \"%s\".
  Use org-mode format with * headings.
  Structure: intro, background, main content with code examples, conclusion." theme)))
           (out-buf (get-buffer-create (format "*draft: %s*" theme))))
      (message "Gemini でブログ下書き生成中...")
      (gptel-request
       prompt
       :buffer out-buf
       :callback
       (lambda (response info)
         (if (not response)
             (message "Gemini エラー: %s" (plist-get info :status))
           (with-current-buffer out-buf
             (org-mode)
             (erase-buffer)
             (insert response))
           (pop-to-buffer out-buf)
           (message "下書きを生成しました"))))))
